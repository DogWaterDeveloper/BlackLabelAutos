// Time picker with restricted minutes (only :00 or :30) function initTimePicker() { const timeInput = document.getElementById('time-input'); // or your time element timeInput.addEventListener('change', function(e) { const timeValue = e.target.value; // "HH:MM" format if (timeValue) { const [hours, minutes] = timeValue.split(':').map(Number); // Round to nearest 30 minutes let roundedMinutes; if (minutes < 15) { roundedMinutes = 0; } else if (minutes < 45) { roundedMinutes = 30; } else { roundedMinutes = 0; // Roll over to next hour const newHours = (hours + 1) % 24; e.target.value = `${String(newHours).padStart(2, '0')}:00`; return; } // Format with leading zeros const formattedTime = `${String(hours).padStart(2, '0')}:${String(roundedMinutes).padStart(2, '0')}`; e.target.value = formattedTime; } }); // Also handle input/blur to prevent invalid entries timeInput.addEventListener('blur', function(e) { const timeValue = e.target.value; if (timeValue) { const [hours, minutes] = timeValue.split(':').map(Number); // Force to either :00 or :30 const validMinutes = minutes < 30 ? 0 : 30; e.target.value = `${String(hours).padStart(2, '0')}:${String(validMinutes).padStart(2, '0')}`; } }); } // Alternative: Use select dropdowns instead for bulletproof restriction function createTimeSelector(containerId) { const container = document.getElementById(containerId); // Hour selector (0-23) const hourSelect = document.createElement('select'); hourSelect.id = 'hour-select'; for (let i = 0; i < 24; i++) { const option = document.createElement('option'); option.value = String(i).padStart(2, '0'); option.textContent = String(i).padStart(2, '0'); hourSelect.appendChild(option); } // Minute selector (only 00 or 30) const minuteSelect = document.createElement('select'); minuteSelect.id = 'minute-select'; [0, 30].forEach(min => { const option = document.createElement('option'); option.value = String(min).padStart(2, '0'); option.textContent = String(min).padStart(2, '0'); minuteSelect.appendChild(option); }); // Separator const separator = document.createElement('span'); separator.textContent = ' : '; separator.style.margin = '0 5px'; container.appendChild(hourSelect); container.appendChild(separator); container.appendChild(minuteSelect); // Get formatted time return { getValue: () => `${hourSelect.value}:${minuteSelect.value}`, setValue: (hours, minutes) => { hourSelect.value = String(hours).padStart(2, '0'); minuteSelect.value = minutes < 30 ? '00' : '30'; } }; }
